        // TB710FU: native touch candidate row
        const purpose = Main.inputMethod.content_purpose;
        const sensitive = purpose === Clutter.InputContentPurpose.PASSWORD ||
            purpose === Clutter.InputContentPurpose.PIN ||
            Main.actionMode === Shell.ActionMode.LOCK_SCREEN;
        const shortcuts = Main.extensionManager.lookup('tb-osk-shortcuts@tb710fu')?.stateObj;
        // TB710FU: stable five-slot candidates, one owner sizes before display.
        if (typeof shortcuts?.setCandidateTable === 'function') {
            const texts = Main.keyboard.visible && !sensitive && this.showLookupTable ? this.table : [];
            shortcuts.setCandidateTable(texts, this._hasPrev, this._hasNext,
                index => {
                    if (this.showLookupTable && this.table[index] === texts[index])
                        this.selectCandidate(index);
                }, direction => this.emit(direction < 0 ? 'LookupTablePageUp' : 'LookupTablePageDown'));
            if (Main.keyboard.visible || sensitive)
                this.inputpanel.hide();
        } else {
            // Preserve native candidates when the optional shared-row helper is disabled.
            Main.keyboard.resetSuggestions();
            if (Main.keyboard.visible) {
                this.inputpanel.hide();
                if (!sensitive && this.showLookupTable && this.table.length) {
                    if (this._hasPrev)
                        Main.keyboard.addSuggestion('‹', () => this.emit('LookupTablePageUp'));
                    this.table.forEach((text, index) => {
                        Main.keyboard.addSuggestion(text, () => {
                            if (this.showLookupTable && this.table[index] === text)
                                this.selectCandidate(index);
                        });
                    });
                    if (this._hasNext)
                        Main.keyboard.addSuggestion('›', () => this.emit('LookupTablePageDown'));
                    Main.keyboard.setSuggestionsVisible(true);
                }
            } else if (sensitive) {
                this.inputpanel.hide();
            }
        }
